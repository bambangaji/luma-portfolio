import { encrypt, decrypt } from './crypto.service';

const BASE = import.meta.env.VITE_API_URL ?? 'http://localhost:3000';

export interface AuthResponse {
  token: string;
  user: { id: string; name: string; email: string };
}

async function post<T>(path: string, body: object): Promise<T> {
  const payload = await encrypt(body);
  const res = await fetch(`${BASE}${path}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ payload }),
  });
  const json = await res.json();
  if (!res.ok) throw new Error(json?.message ?? 'Request failed');
  // Server returns { payload: '<encrypted>' }
  return decrypt<T>(json.payload);
}

export const AuthService = {
  login: (email: string, password: string) =>
    post<AuthResponse>('/auth/login', { email, password }),

  register: (name: string, email: string, password: string) =>
    post<AuthResponse>('/auth/register', { name, email, password }),
};

                attempt++
            ) {
                try {
                    const response = await fetch(
                        this.URL,
                        {
                            method: 'POST',
                            headers: {
                                'Content-Type':
                                    'application/json',
                            },
                            body: JSON.stringify({
                                resume_text: JSON.stringify(
                                    resumeText.sections
                                ),
                                template_json:
                                    templateJson,
                            }),
                        }
                    );

                    if (!response.ok) {
                        throw new Error(
                            `Request failed (${response.status})`
                        );
                    }

                    const result =
                        await response.json();

                    // Retry if backend says failed
                    if (
                        result?.success ===
                        false
                    ) {
                        throw new Error(
                            result.message ||
                            result.error ||
                            'Generation failed'
                        );
                    }

                    toast.success(
                        `Resume generated successfully`,
                        {
                            id: toastId,
                        }
                    );

                    lastError =
                        error instanceof Error

                            : new Error(
                                'Unknown error'
                            );

import { createSlice, createAsyncThunk } from "@reduxjs/toolkit";
import type { PayloadAction } from "@reduxjs/toolkit";
import { userService } from "../services/user-service";
import type { User } from "../interface/user-model";


export const fetchUsers = createAsyncThunk("users/fetch", async () => {
  return await userService.fetchUsers();
});

interface UserState {
  users: User[];
  loading: boolean;
  error: string | null;
}

const initialState: UserState = {
  users: [],
  loading: false,
  error: null,
};

const userSlice = createSlice({
  name: "user",
  initialState,
  reducers: {},
  extraReducers: (builder) => {
    builder
      .addCase(fetchUsers.pending, (state) => {
        state.loading = true;
      })
      .addCase(fetchUsers.fulfilled, (state, action: PayloadAction<User[]>) => {
        state.loading = false;
        state.users = action.payload;
      })
      .addCase(fetchUsers.rejected, (state, action) => {
        state.loading = false;
        state.error = action.error.message || "Something went wrong";
      });
  },
});

export default userSlice.reducer;

## Overview

**LUMA PORTFOLIO** is a React-based Single Page Application (SPA) that enables users to generate professional portfolio websites directly from their resume/CV. The core workflow is:

1. Upload a PDF resume
2. AI parses and classifies the resume content
3. A structured portfolio is generated automatically
4. User can preview, edit, and customize the portfolio via a CMS-style editor
5. Portfolio is ready to share or publish

The platform supports multiple professional fields including Photography, Accounting, Design, Engineering, Healthcare, Education, and more.

import type { PortfolioJson } from "../interface/portofolio-json.interface";
import type { Experience, Project, ResumeJson } from "../interface/resume-json.interface";

export class PortfolioMappingService {
  static mapResumeToPortfolio(
    resume: ResumeJson
  ): PortfolioJson {
    const latestExperience =
      resume.experiences.find(
        experience =>
          experience.is_current
      ) ??
      resume.experiences[0];

    const totalSkills =
      resume.skills.reduce(
        (
          total,
          category
        ) =>
          total +
          category.items.length,
        0
      );

    const totalCompanies =
      new Set(
        resume.experiences.map(
          experience =>
            experience.company
        )
      ).size;

    const totalProjects =
      resume.projects.length;

    const yearsExperience =
      this.calculateYearsExperience(
        resume.experiences
      );

    return {
      hero: {
        badge:
          latestExperience?.role ??
          '',

        headline:
          latestExperience?.company ??
          '',

        title: {
          prefix: '',
          highlight: '',
        },

        description: '',
      },

      skill_section: {
        title:
          'Skills & Expertise',

        subtitle: '',

        categories:
          resume.skills.map(
            skill => ({
              category:
                skill.category,

              score:
                this.calculateSkillScore(
                  skill.items.length
                ),

              items:
                skill.items,
            })
          ),
      },

      experience_section: {
        title:
          'Work Experience',

        items:
          resume.experiences,
      },

      project_section: {
        title:
          'Featured Projects',

        categories:
          this.extractProjectCategories(
            resume.projects
          ),

        items:
          resume.projects,
      },

      stats: [
        {
          label:
            'Years Experience',
          value:
            yearsExperience,
        },

        {
          label:
            'Projects',
          value:
            totalProjects,
        },

        {
          label:
            'Companies',
          value:
            totalCompanies,
        },

        {
          label:
            'Skills',
          value:
            totalSkills,
        },
      ],
      education_section:
        resume.education.map(
          education => ({
            degree:
              education.degree,
            institution:
              education.institution,
            gpa:
              education.gpa,
            start_date:
              education.start_date,
            end_date:
              education.end_date,
            description:
              education.description,
          })
        ),
    }
  }
  private static calculateSkillScore(
      count: number
    ): number {
    if (count <= 3)
      return 60;

    if (count <= 6)
      return 75;

    if (count <= 10)
      return 90;

    return 100;
  }

  private static calculateYearsExperience(
    experiences: Experience[]
  ): number {
    const dates =
      experiences
        .map(
          experience =>
            Date.parse(
              experience.start_date
            )
        )
        .filter(
          date =>
            !Number.isNaN(
              date
            )
        );

    if (
      dates.length === 0
    ) {
      return 0;
    }

    const earliest =
      Math.min(...dates);

    return (
      new Date().getFullYear() -
      new Date(
        earliest
      ).getFullYear()
    );
  }

  private static extractProjectCategories(
    projects: Project[]
  ): string[] {
    const categories =
      new Set<string>();

    const technologyToCategory:
      Record<
        string,
        string
      > = {
      Angular:
        'Frontend',

      React:
        'Frontend',

      Vue:
        'Frontend',

      'Spring Boot':
        'Backend',

      NodeJS:
        'Backend',

      '.NET':
        'Backend',

      Flutter:
        'Mobile',

      Ionic:
        'Mobile',

      PostgreSQL:
        'Database',

      MySQL:
        'Database',

      AWS:
        'Cloud',

      Docker:
        'Cloud',

      Kubernetes:
        'Cloud',

      n8n: 'AI',

      LLaMA:
        'AI',
    };

    projects.forEach(
      project => {
        project.technologies.forEach(
          (technology: string) => {
            categories.add(
              technologyToCategory[
              technology
              ] ??
              technology
            );
          }
        );
      }
    );

    return [
      ...categories,
    ];
  }
}

import type { ResumeJson } from "../interface/resume-json.interface";
import { PortfolioMappingService } from "./portofolio-mapping-service";

export class PortfolioMergeService {
    static merge(
        resume: ResumeJson,
        aiPortfolio: PortfolioJson
    ): PortfolioJson {
        const mapped =
            PortfolioMappingService.mapResumeToPortfolio(
                resume
            );

        return {
            hero: {
                badge:
                    mapped.hero.badge,

                headline:
                    aiPortfolio.hero
                        ?.headline ??
                    '',

                title: {
                    prefix:
                        aiPortfolio.hero
                            ?.title
                            ?.prefix ??
                        '',

                    highlight:
                        aiPortfolio.hero
                            ?.title
                            ?.highlight ??
                        '',
                },

                description:
                    resume
                        .personal_information
                        ?.description ??
                    '',
                short_description:
                    aiPortfolio.hero
                        ?.description ??
                    '',
            },

            skill_section: {
                title:
                    mapped.skill_section
                        .title,

                subtitle:
                    aiPortfolio
                        .skill_section
                        ?.subtitle ??
                    '',

                categories:
                    mapped.skill_section
                        .categories,
            },

            experience_section:
                mapped.experience_section,

            project_section:
                mapped.project_section,

            stats:
