import type { PortfolioJson } from "../interface/portofolio-json.interface";
import type { Experience, Project, ResumeJson } from "../interface/resume-json.interface";

export class PortfolioMappingService {
  static mapResumeToPortfolio(
    resume: ResumeJson
  ): PortfolioJson {
    const latestExperience =
      resume.experiences.find(
        experience =>
          experience.is_current
      ) ??
      resume.experiences[0];

    const totalSkills =
      resume.skills.reduce(
        (
          total,
          category
        ) =>
          total +
          category.items.length,
        0
      );

    const totalCompanies =
      new Set(
        resume.experiences.map(
          experience =>
            experience.company
        )
      ).size;

    const totalProjects =
      resume.projects.length;

    const yearsExperience =
      this.calculateYearsExperience(
        resume.experiences
      );

    return {
      hero: {
        badge:
          latestExperience?.role ??
          '',

        headline:
          latestExperience?.company ??
          '',

        title: {
          prefix: '',
          highlight: '',
        },

        description: '',
      },

      skill_section: {
        title:
          'Skills & Expertise',

        subtitle: '',

        categories:
          resume.skills.map(
            skill => ({
              category:
                skill.category,

              score:
                this.calculateSkillScore(
                  skill.items.length
                ),

              items:
                skill.items,
            })
          ),
      },

      experience_section: {
        title:
          'Work Experience',

        items:
          resume.experiences,
      },

      project_section: {
        title:
          'Featured Projects',

        categories:
          this.extractProjectCategories(
            resume.projects
          ),

        items:
          resume.projects,
      },

      stats: [
        {
          label:
            'Years Experience',
          value:
            yearsExperience,
        },

        {
          label:
            'Projects',
          value:
            totalProjects,
        },

        {
          label:
            'Companies',
          value:
            totalCompanies,
        },

        {
          label:
            'Skills',
          value:
            totalSkills,
        },
      ],
      education_section:
        resume.education.map(
          education => ({
            degree:
              education.degree,
            institution:
              education.institution,
            gpa:
              education.gpa,
            start_date:
              education.start_date,
            end_date:
              education.end_date,
            description:
              education.description,
          })
        ),
    }
  }
  private static calculateSkillScore(
      count: number
    ): number {
    if (count <= 3)
      return 60;

    if (count <= 6)
      return 75;

    if (count <= 10)
      return 90;

    return 100;
  }

  private static calculateYearsExperience(
    experiences: Experience[]
  ): number {
    const dates =
      experiences
        .map(
          experience =>
            Date.parse(
              experience.start_date
            )
        )
        .filter(
          date =>
            !Number.isNaN(
              date
            )
        );

    if (
      dates.length === 0
    ) {
      return 0;
    }

    const earliest =
      Math.min(...dates);

    return (
      new Date().getFullYear() -
      new Date(
        earliest
      ).getFullYear()
    );
  }

  private static extractProjectCategories(
    projects: Project[]
  ): string[] {
    const categories =
      new Set<string>();

    const technologyToCategory:
      Record<
        string,
        string
      > = {
      Angular:
        'Frontend',

      React:
        'Frontend',

      Vue:
        'Frontend',

      'Spring Boot':
        'Backend',

      NodeJS:
        'Backend',

      '.NET':
        'Backend',

      Flutter:
        'Mobile',

      Ionic:
        'Mobile',

      PostgreSQL:
        'Database',

      MySQL:
        'Database',

      AWS:
        'Cloud',

      Docker:
        'Cloud',

      Kubernetes:
        'Cloud',

      n8n: 'AI',

      LLaMA:
        'AI',
    };

    projects.forEach(
      project => {
        project.technologies.forEach(
          (technology: string) => {
            categories.add(
              technologyToCategory[
              technology
              ] ??
              technology
            );
          }
        );
      }
    );

    return [
      ...categories,
    ];
  }
}

## Overview

**LUMA PORTFOLIO** is a React-based Single Page Application (SPA) that enables users to generate professional portfolio websites directly from their resume/CV. The core workflow is:

1. Upload a PDF resume
2. AI parses and classifies the resume content
3. A structured portfolio is generated automatically
4. User can preview, edit, and customize the portfolio via a CMS-style editor
5. Portfolio is ready to share or publish

The platform supports multiple professional fields including Photography, Accounting, Design, Engineering, Healthcare, Education, and more.

---

| Feature | Description |
|---|---|
| 🤖 AI Resume Parsing | Upload a PDF resume; AI extracts and classifies structured data |
| 🎨 Portfolio Builder | CMS-style editor with live preview |
| 🌗 Dark / Light Mode | Global theme toggle with smooth transitions |
| 🌐 Multi-language | i18n support with language toggle |
| 📊 Skills Charts | Circle/radar charts for skill visualization |
| 📅 Date Picker | Calendar-based date picker for experience & education |
| 🔐 Authentication | Login, Register, Forgot Password flows |
| 👁️ Live Preview | Real-time portfolio preview while editing |
| 🖼️ Thumbnail View | Portfolio thumbnail previews |

The `n8n/` directory contains automation workflow JSONs used for AI-powered resume processing:

| File | Purpose |
|---|---|
| `classification-resume.json` | Classifies resume content into structured categories |
| `generate-resume.json` | Generates portfolio data from classified resume |

These workflows are designed to be imported into an [n8n](https://n8n.io) instance and connected to the frontend via API calls.
